cmake_minimum_required(VERSION 3.24)
project(iiNetworkConsumer LANGUAGES CXX)
find_package(iiNetwork 0.4 CONFIG REQUIRED)
if(NOT iiNetwork_REALTIME_LATEST)
    message(FATAL_ERROR "This consumer requires the iiNetwork package with native realtime channels")
endif()
add_executable(iiNetworkInstalledRealtime ../realtime.cpp)
target_link_libraries(iiNetworkInstalledRealtime PRIVATE iiNetwork::iiNetwork)
add_executable(iiNetworkInstalledRealtimeExample ../../example/realtime.cpp)
target_link_libraries(iiNetworkInstalledRealtimeExample PRIVATE iiNetwork::iiNetwork)
add_executable(iiNetworkConsumer ../../examples/roundtrip.cpp)
target_link_libraries(iiNetworkConsumer PRIVATE iiNetwork::iiNetwork)
enable_testing()
add_executable(iiNetworkInstalledLoadBalancer ../load_balancer.cpp)
target_link_libraries(iiNetworkInstalledLoadBalancer PRIVATE iiNetwork::iiNetwork)
add_test(NAME iiNetwork.Installed.LoadBalancer COMMAND iiNetworkInstalledLoadBalancer)
set_tests_properties(iiNetwork.Installed.LoadBalancer PROPERTIES TIMEOUT 30)
foreach(_case rooms ordering bounds authentication lifecycle lifetime negotiation pacing feedback reentry protocol rtt)
    add_test(NAME iiNetwork.Installed.realtime.${_case} COMMAND iiNetworkInstalledRealtime ${_case})
    set_tests_properties(iiNetwork.Installed.realtime.${_case} PROPERTIES TIMEOUT 30)
endforeach()
add_test(NAME iiNetwork.InstalledRealtimeExample COMMAND iiNetworkInstalledRealtimeExample)
set_tests_properties(iiNetwork.InstalledRealtimeExample PROPERTIES TIMEOUT 30)
add_test(NAME iiNetwork.InstalledConsumer COMMAND iiNetworkConsumer)
add_executable(iiNetworkInstalledProtocols protocols.cpp)
target_link_libraries(iiNetworkInstalledProtocols PRIVATE iiNetwork::iiNetwork)
add_test(NAME iiNetwork.InstalledProtocols COMMAND iiNetworkInstalledProtocols)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
add_executable(iiNetworkInstalledRealtimeProcess ../realtime_process.cpp)
target_link_libraries(iiNetworkInstalledRealtimeProcess PRIVATE iiNetwork::iiNetwork)
add_test(NAME iiNetwork.Installed.realtime.process COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../realtime_process.py" "$<TARGET_FILE:iiNetworkInstalledRealtimeProcess>")
set_tests_properties(iiNetwork.Installed.realtime.process PROPERTIES TIMEOUT 60)
get_filename_component(_openssl_prefix "${OPENSSL_INCLUDE_DIR}" DIRECTORY)
find_program(IINETWORK_OPENSSL_COMMAND NAMES openssl HINTS "${_openssl_prefix}/bin" "${_openssl_prefix}/tools/openssl" REQUIRED)
add_executable(iiNetworkInstalledInterfaces ../interfaces.cpp)
target_link_libraries(iiNetworkInstalledInterfaces PRIVATE iiNetwork::iiNetwork)
add_test(NAME iiNetwork.InstalledTlsFixtures COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../create_tls_fixtures.py" "${IINETWORK_OPENSSL_COMMAND}" "${CMAKE_BINARY_DIR}/fixtures")
set_tests_properties(iiNetwork.InstalledTlsFixtures PROPERTIES FIXTURES_SETUP iiNetworkInstalledTls TIMEOUT 60)
foreach(_case address tcp udp local tls http servers)
    add_test(NAME iiNetwork.Installed.${_case} COMMAND iiNetworkInstalledInterfaces ${_case} "${CMAKE_BINARY_DIR}/fixtures")
    set_tests_properties(iiNetwork.Installed.${_case} PROPERTIES FIXTURES_REQUIRED iiNetworkInstalledTls TIMEOUT 30)
endforeach()
add_executable(iiNetworkInstalledPeers ../peers.cpp)
target_link_libraries(iiNetworkInstalledPeers PRIVATE iiNetwork::iiNetwork)
foreach(_case duplex lifecycle policy protocol transports)
    add_test(NAME iiNetwork.Installed.peers.${_case} COMMAND iiNetworkInstalledPeers ${_case} "${CMAKE_BINARY_DIR}/fixtures")
    set_tests_properties(iiNetwork.Installed.peers.${_case} PROPERTIES TIMEOUT 30)
endforeach()
set_tests_properties(iiNetwork.Installed.peers.transports PROPERTIES FIXTURES_REQUIRED iiNetworkInstalledTls)
add_executable(iiNetworkInstalledPeerExample ../../examples/peers.cpp)
target_link_libraries(iiNetworkInstalledPeerExample PRIVATE iiNetwork::iiNetwork)
add_test(NAME iiNetwork.InstalledPeerExample COMMAND iiNetworkInstalledPeerExample)
