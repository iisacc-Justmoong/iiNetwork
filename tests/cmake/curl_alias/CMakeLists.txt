cmake_minimum_required(VERSION 3.24)
project(CurlIncludeContracts LANGUAGES NONE)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/AppleCurlIncludes.cmake")
add_library(Plain INTERFACE IMPORTED)
add_library(Actual INTERFACE IMPORTED)
add_library(CurlAlias ALIAS Actual)
foreach(target IN ITEMS Plain Actual)
    set_property(TARGET "${target}" PROPERTY INTERFACE_INCLUDE_DIRECTORIES "/kit/include;/sdk/usr/include")
endforeach()
iinetwork_normalize_apple_curl(Plain /sdk)
iinetwork_normalize_apple_curl(CurlAlias /sdk)
foreach(target IN ITEMS Plain Actual)
    get_target_property(actual "${target}" INTERFACE_INCLUDE_DIRECTORIES)
    if(NOT actual STREQUAL "/kit/include")
        message(FATAL_ERROR "${target} lost the kit includes or retained the system C include override")
    endif()
endforeach()
