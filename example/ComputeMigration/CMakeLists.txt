cmake_minimum_required(VERSION 3.31)
project(iiSDKComputeMigration LANGUAGES CXX)
include(CTest)
find_package(iiSDKCore 0.20 CONFIG REQUIRED COMPONENTS Runtime)
find_package(iiNetwork 0.1 CONFIG REQUIRED)
add_library(iiSDKRemoteWork App/RemoteWork.cpp)
target_compile_features(iiSDKRemoteWork PUBLIC cxx_std_23)
target_include_directories(iiSDKRemoteWork PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/App")
target_link_libraries(iiSDKRemoteWork PUBLIC iiSDKCore::Runtime iiNetwork::iiNetwork)
if(BUILD_TESTING)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    add_executable(iiSDKRemoteWorkTests tests/remote_work.cpp)
    target_link_libraries(iiSDKRemoteWorkTests PRIVATE iiSDKRemoteWork)
    add_test(NAME iiSDK.RemoteWork COMMAND iiSDKRemoteWorkTests)
    set_tests_properties(iiSDK.RemoteWork PROPERTIES TIMEOUT 30)
    add_test(NAME iiSDK.RemoteWorkProcesses COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/process_transfer.py" "$<TARGET_FILE:iiSDKRemoteWorkTests>")
    set_tests_properties(iiSDK.RemoteWorkProcesses PROPERTIES TIMEOUT 45)
    find_package(iiGPU 0.1 CONFIG QUIET)
    if(TARGET iiGPU::iiGPU)
        add_executable(iiSDKCpuGpuMigrationTests tests/cpu_gpu.cpp)
        target_compile_features(iiSDKCpuGpuMigrationTests PRIVATE cxx_std_23)
        target_include_directories(iiSDKCpuGpuMigrationTests PRIVATE "${iiGPU_EXAMPLE_DIR}")
        target_link_libraries(iiSDKCpuGpuMigrationTests PRIVATE iiSDKCore::Runtime iiGPU::iiGPU)
        add_test(NAME iiSDK.CpuGpuMigration COMMAND iiSDKCpuGpuMigrationTests)
        set_tests_properties(iiSDK.CpuGpuMigration PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    endif()
endif()
